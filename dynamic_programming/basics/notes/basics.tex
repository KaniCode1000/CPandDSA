	\section{Maximum Weight Independent Set on a Path Graph}
	
	\textbf{Independent Set:} A subset of vertices $S \subseteq V$ such that no two vertices in $S$ are adjacent. For path graph, this would mean if $v_i \in S \implies \{v_{i-1}, v_{i+1}\} \cap S = \emptyset$
	\vspace{3mm}
	
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Problem: MWIS in path graph}
		\KwIn{An path graph $G = (V,E)$ with $n$ vertices, denoted $v_1,v_2, \ldots ,v_n$, each having non negative weight $w_i$}
		\KwOut{Find an independent set where sum of weights is maximised, also find its sum}
	\end{algorithm}
	\vspace{5mm}
	
	Let $A[i]$ be the maximum weight for the first $i$ vertices. The Recursion becomes:
	\[A[i] = max(A[i-1],A[i-2] + w_i)\]
	
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Solution}
		$A[0] \leftarrow 0, A[1] \leftarrow w[1]$ \;
		\For{$i \leftarrow 2$ to $n$ }{
			$A[i] = max(A[i-1],A[i-2] + w_i)$
		}
		\Return{$A[n]$}
	\end{algorithm}
	\vspace{2mm}
	Go backward from $n$ to reconstruct the set of vertices chosen.
	
	\section{Weighted Interval Scheduling}
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Problem: Weighted Interval Scheduling}
		\KwIn{set of jobs, with each job having a start time, finish time, and a weight (value/profit). }
		\KwOut{A subset of non-overlapping jobs that maximizes the total weight.}
	\end{algorithm}	
	\vspace{5mm}
	\begin{itemize}
		\item Let the $n$ intervals be sorted by finish times.
		\item $OPT[i]$ be the maximum weight for the first $i$ intervals or jobs.
		\item $p(i)$ be the largest index $j < i$ such that $f_j \leq s_i$.
	\end{itemize}
	The Recursion becomes:
	\[OPT[i] = max(OPT[i-1], v_i + OPT[p[(i)])\]
	
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Solution}
		$OPT[0] \leftarrow 0$ \, //of size n+1 \;
		\For{$i \leftarrow 1$ to $n$ }{
			$OPT[i] = max(OPT[i-1], v_i + OPT[p[(i)])$
		}
		\Return{$OPT[n]$}
	\end{algorithm}
	\vspace{2mm}
	Here, the values $p(i)$ have been precomputed using binary search, and the sorting has been done beforehand.
	
	\section{0/1 Knapsack Problem}
	
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Problem: 0/1 Knapsack}
		\KwIn{There are $n$ items with weight $w_i$, value $v_i$ and $W$ is the maximum weight that the knapsack can hold.}
		\KwOut{Find the maximum value that the knapsack can hold.}
	\end{algorithm}
	\vspace{5mm}
	
	Let $K[i,w]$ be the maximum value achievable using the first $i$ items with a combined weight of $w_i$. The Recursion becomes:
	\[
	K[i,w] =
	\begin{cases}
		K[i-1,w] & \text{if } w_i > w, \\[6pt]
		\max\big(K[i-1,w], \; v_i + K[i-1,w-w_i]\big) & \text{if } w_i \le w.
	\end{cases}
	\]
	
	Here $w \in [0,W]$ and K is a $2D$ array of size $(n+1) \times (W+1)$
	\vspace{2mm}
	
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Solution}
		\For{$w \leftarrow 0$ to $W$ }{
			$K[0,w] \leftarrow 0$
		} \;
		\For{$i \leftarrow 0$ to $n$ }{
			$K[i,0] \leftarrow 0$
		} \;

	\For{$i \gets 1$ \KwTo $n$}{ % For each item
		\For{$j \gets 1$ \KwTo $W$}{ % For each capacity
			\If{$w[i] > j$}{
				$K[i,j] \gets K[i-1,j]$\;
			}
			\Else{
				$K[i,j] \gets \max(K[i-1,j], v[i] + K[i-1, j - w[i]])$\;
			}
		}
	}
		
		\Return{$K[n,W]$}
	\end{algorithm}
	
	\section{Longest Increasing Subsequence}
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Problem: LIS}
		\KwIn{Array $a$,with $n$ numbers}
		\KwOut{Find the longest strictly increasing subsequence in $a$}
	\end{algorithm}
	\vspace{5mm}
	
	\subsection{Solution in $O(n^2)$}
	Define $d[0,1, \ldots n],$ where $d[i]$ is the length of the longest increasing subsequence ending at index $i$. Therefore,
	\[
	d[i] = \max \left(1,
	\underset{\substack{
			j < i \\
			a[j] < a[i]
	}}{\max} (d[j] + 1) \right)
	\]
	maintaining an array $p[i] = j$, where $j$ is the maximal value of $d[j]$ found for index $i$, also allows us to reconstruct the sequence (Not required but is the easier way).
	
	\subsection{Solution in $O(n \, log \, n)$}
	We define $d[0, \dots n]$ such that $d[l]$ defines the smallest element at which an increasing subsequence of length $l$ ends. \newline
	
	We go from left to right and find the place to insert $a[i]$ in $d$ using binary search. Hence, the Time Complexity - $O(n \, log \, n)$ and Space Complexity - $O(n)$.
	
	\subsubsection{Restoring subsequence}
	We maintain two auxiliary arrays, one that tells us the index of elements in $d[]$, and an array of ancestors $p[]$ such that $p[i]$ will be the index of the previous element for the optimal subsequence ending at element $i$.
	
	\section{Coin Change (Ways)}
	\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Problem: Coin Change (Ways)}
		\KwIn{Denominations of coins in array $A$ and the target amount $N$}
		\KwOut{Number of ways to achieve the target amount using denomination of coins such that the order does not matter.}
	\end{algorithm}
	\vspace{5mm}
	
	\subsection{Solution in $O(N \cdot K)$}
	
	Idea is simple, consider a $dp[i][j]$ such that it stores the number of ways
	to achieve value $i$ using the last coin as $A[x]$ where $x < j \land A[x] < A[j]$. Since, the order does not matter, we are counting only the lexicographical order. Now, from coin $j$, we can either use the coin $j$ again, or move onto obtaining the value $i$ using coin $j+1$.
	
	\subsection{Transition}
	\[
		dp[i,j] = dp[i, j-1] + dp[i-A[j], \,j] 
	\]
	
	\section{Subset Sum with Add and Erase}
		\begin{algorithm}[H]
		\SetAlgoLined
		\caption{Problem: Subset Sum}
		\KwIn{An array $A$ and value $K$}
		\KwOut{Number of subsets of $A$ with sum $K$}
	\end{algorithm}
	\vspace{5mm}
	
	\textbf{Subset:} Set of indices (not values).
	
	\subsection{Solution in $O(N \cdot K)$}
	
	$dp[i] = $ Number of subsets whose sum is $i$. Base Case: $dp[0] = 1$ (the empty subset).
	
	\subsection{Transition}
	
	\[
	dp[i] = dp[i] + dp[i-x] \quad \forall x \leq i
	\]
	
	\subsection{Add and Erase}
	Start with an empty array, and follow $Q$ queries of form either $+x$ or $-x$.
	\subsubsection{Addition}
	For a query $+x$, transitions becomes:
	\[
	\forall K, K-1 \ldots x, \, \text{in this order do:} \quad dp[i] = dp[i] + dp[i-x] 
	\]
	
	\subsubsection{Erasing}
	For a query $-x$, transitions becomes:
	\[
	\forall x, x+1 \ldots K, \, \text{in this order do:} \quad dp[i] = dp[i] - dp[i-x] 
	\]
	
	\section{Levenshtein Distance}
		\begin{algorithm}[H]
			\SetAlgoLined
			\caption{Problem: Levenshtein Distance}
			\KwIn{Two strings $x$ and $y$}
			\KwOut{Minimum number of edits required to convert $x$ into $y$, operations allowed are insert (in $x$), remove (from $x$) and modify (a character of $x$) which cost of each operation being $1$.}
		\end{algorithm}
		\vspace{5mm}
	
	Idea is simple, suppose $dp[a,b]$ denotes the minimum number of edits required to match $x[0, \ldots a]$ and $y[0, \ldots b]$
	
	\[
		dp[a,b] = \min \left(dp[a-1,b] + 1, dp[a,b-1] + 1, dp[a-1,b-1] + cost(a,b)\right)	
	\]
	
	where $cost(a,b)$ is defined as:
	
	\[
	cost(a,b) = \begin{cases}
		x[a] = y[b] & 0, \\
		x[a] \neq y[b] & 1
	\end{cases}
	\]
	
	Here, matching of string till $a$ and $b$ means either inserting at position $a \, (dp[a,b-1])$, removing at position $a \, (dp[a-1,b])$ or modifying (might not need to if $x[a] = y[b]$) at position $a \, (dp[a-1,b-1])$.
